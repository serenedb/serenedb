using System;
using System.Collections.Generic;
using Microsoft.EntityFrameworkCore;

namespace Scaffolded;

public partial class ScaffoldedContext : DbContext
{
    public ScaffoldedContext(DbContextOptions<ScaffoldedContext> options)
        : base(options)
    {
    }

    public virtual DbSet<Blog> Blogs { get; set; }

    public virtual DbSet<Comment> Comments { get; set; }

    public virtual DbSet<Post> Posts { get; set; }

    public virtual DbSet<Tag> Tags { get; set; }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.Entity<Blog>(entity =>
        {
            entity.ToTable(tb => tb.HasComment("blogs we follow"));

            entity.HasIndex(e => e.Name, "IX_Blogs_Name")
                .IsUnique()
                .HasMethod("secondary");

            entity.Property(e => e.Id).HasDefaultValueSql("nextval('Blogs_Id_seq')");
            entity.Property(e => e.Active).HasDefaultValue(true);
            entity.Property(e => e.CreatedAt).HasDefaultValueSql("now()");
            entity.Property(e => e.OrderNumber).HasDefaultValueSql("nextval('blog_numbers')");
            entity.Property(e => e.Url).HasComment("home page");
        });

        modelBuilder.Entity<Comment>(entity =>
        {
            entity.HasIndex(e => new { e.PostId, e.At }, "IX_Comments_PostId_At").HasMethod("secondary");

            entity.Property(e => e.Id).HasDefaultValueSql("nextval('Comments_Id_seq')");

            entity.HasOne(d => d.Post).WithMany(p => p.Comments)
                .HasForeignKey(d => d.PostId)
                .OnDelete(DeleteBehavior.ClientSetNull);
        });

        modelBuilder.Entity<Post>(entity =>
        {
            entity.HasIndex(e => new { e.BlogId, e.Published }, "IX_Posts_BlogId_Published").HasMethod("secondary");

            entity.HasIndex(e => e.ExternalId, "IX_Posts_ExternalId")
                .IsUnique()
                .HasMethod("secondary");

            entity.HasIndex(e => e.Score, "IX_Posts_Score").HasMethod("secondary");

            entity.Property(e => e.Id).HasDefaultValueSql("nextval('Posts_Id_seq')");
            entity.Property(e => e.Rating).HasPrecision(5, 2);

            entity.HasOne(d => d.Blog).WithMany(p => p.Posts)
                .HasForeignKey(d => d.BlogId)
                .OnDelete(DeleteBehavior.ClientSetNull);

            entity.HasMany(d => d.Tags).WithMany(p => p.Posts)
                .UsingEntity<Dictionary<string, object>>(
                    "PostTag",
                    r => r.HasOne<Tag>().WithMany()
                        .HasForeignKey("TagsId")
                        .OnDelete(DeleteBehavior.ClientSetNull),
                    l => l.HasOne<Post>().WithMany()
                        .HasForeignKey("PostsId")
                        .OnDelete(DeleteBehavior.ClientSetNull),
                    j =>
                    {
                        j.HasKey("PostsId", "TagsId");
                        j.ToTable("PostTag");
                        j.HasIndex(new[] { "TagsId" }, "IX_PostTag_TagsId").HasMethod("secondary");
                    });
        });

        modelBuilder.Entity<Tag>(entity =>
        {
            entity.HasIndex(e => e.Label, "IX_Tags_Label")
                .IsUnique()
                .HasMethod("secondary");

            entity.Property(e => e.Id).HasDefaultValueSql("nextval('Tags_Id_seq')");
        });
        modelBuilder.HasSequence("blog_numbers")
            .StartsAt(1000L)
            .IncrementsBy(5)
            .HasMax(2147483647L);

        OnModelCreatingPartial(modelBuilder);
    }

    partial void OnModelCreatingPartial(ModelBuilder modelBuilder);
}
